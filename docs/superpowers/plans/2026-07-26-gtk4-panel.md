# GTK4 Panel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace CCAS's chain of fuzzel windows with one GTK4 layer-shell panel — header with account chips and usage bars, launch verbs, a two-pane project/session browser with live search, and inline toggles.

**Architecture:** `ccas/panel.py` is pure Python: it turns registry + usage + history into a plain state dict and filters it. `ccas/panel_ui.py` is the only module importing `gi`; it renders that dict and returns an `Action`. `ccas/cli.py` dispatches the action through the existing `launch` / `registry` / `accounts` entry points. No decision is made inside the widget tree.

**Tech Stack:** Python 3.14, PyGObject 3.56, GTK 4.22, gtk4-layer-shell 1.3, pytest.

Design: `docs/superpowers/specs/2026-07-26-gtk4-panel-design.md`. Read it before Task 1.

## Global Constraints

- **Never write to `~/.claude`.** CCAS only reads it. `test_relink_never_touches_mtimes_in_claude_home` is the canary.
- **Never delete.** Everything moves to `~/.claude_trash/`. This includes `install.sh`.
- **Never invoke `claude` by bare name.** Always `paths.claude_bin()`.
- **`paths.py` owns every filesystem location**, each `CCAS_*`-overridable. No other module may hardcode a path.
- **`gi` is imported inside `panel_ui.py` only, and lazily** (inside functions, not at module top). `ccs -p`, `ccs list`, `ccs doctor`, `ccs statusline` must keep working with PyGObject absent.
- **`ccas/panel.py` must never import `gi`**, directly or transitively.
- **Tests must never touch real state.** Use the `CCAS_*` env overrides via tmp_path.
- **Never co-sign or co-author commits.** No `Co-Authored-By`, no trailers.
- **`ccs` runs the installed copy.** Re-run `./install.sh` before every live check.
- Python 3.14, stdlib only at runtime outside `panel_ui.py`. No build step.
- Test suite is `python -m pytest` from the repo root, ~254 tests, under a second.
- One test file per module, same name.

```python
"""The panel's state: everything decided before a widget exists.

Deliberately free of GTK. build_state() turns the registry, the usage reading
and the session history into a plain dict; filter_state() narrows it. The
widget tree in panel_ui.py renders that dict and decides nothing, which is what
keeps the panel's behaviour inside the test suite rather than inside a toolkit.

test_panel_never_imports_gi pins the split.
"""
import time
from collections import namedtuple

from . import history, label, paths, registry, usage

# What the widget tree hands back. `kind` says which branch cli.py dispatches;
# `value` is that branch's argument and is None for the kinds that take none.
Action = namedtuple("Action", "kind slug value")

WINDOW_LABELS = {"five_hour": "5h", "seven_day": "wk"}


def _usage_rows(slug: str, now: float):
    reading = usage.read(slug)
    rows = []
    for key in usage.WINDOWS:
        st = usage.state(reading, key, now)
        rows.append({
            "key": key,
            "label": WINDOW_LABELS[key],
            "kind": st.kind,
            "percent": st.percent,
            "resets": "" if st.resets_at is None
                      else usage.reset_time(st.resets_at, now),
            "color": None if st.percent is None else usage.color(st.percent),
        })
    return rows


def _accounts(reg: dict, slug: str):
    return [{
        "slug": a["slug"],
        "name": label.display_name(a),
        "email": a.get("email") or "",
        "color": paths.PALETTE[a["color"]][1],
        "current": a["slug"] == slug,
    } for a in reg["accounts"]]


def build_state(slug: str, now=None) -> dict:
    now = time.time() if now is None else now
    reg = registry.load()
    account = registry.find(reg, slug) or {}
    return {
        "slug": slug,
        "accounts": _accounts(reg, slug),
        "display": account.get("display", "nickname"),
        "hide_icon": bool(account.get("hide_icon")),
        "headless": bool(account.get("headless")),
        "dangerous": bool(account.get("dangerous")),
        "usage": _usage_rows(slug, now),
    }
```

```python
UNTITLED = "(untitled)"


def _panes(now: float):
    """(projects, sessions-by-project). Ordered by mtime throughout.

    history.scan() is already mtime-descending and project_dirs() preserves
    that order while dropping directories that no longer exist, so the project
    column falls out of it — the first project is the one with the most recent

    by_project = {}
    for s in sessions:

            "uuid": s.uuid,
            "title": s.title or UNTITLED,
            "age": history.humanise_age(now - s.mtime),
            "mtime": s.mtime,
            "cwd": s.cwd,

        "path": d,
        "short": history.abbreviate(d),
        "age": history.humanise_age(now - by_project[d][0]["mtime"]),
        "mtime": by_project[d][0]["mtime"],
    } for d in history.project_dirs(sessions) if d in by_project]
    return projects, by_project
```

```python
def filter_state(state: dict, query: str) -> dict:
    """Narrow both panes to `query`. A new dict; the argument is untouched.

    Both panes, not just the sessions: the search earns its place when you
    cannot remember which project something was in, and a left column that
    stays full while the right one empties says the opposite.

    A project whose *path* matches keeps all of its sessions, so typing a
    directory name works as a project filter rather than as a filter that
    finds the project and then hides everything in it.
    """
    q = query.strip().lower()
    if not q:
        return state

    sessions, projects = {}, []
    for project in state["projects"]:
        rows = state["sessions"].get(project["path"], [])
        if q in project["short"].lower():
            hits = rows
        else:
            hits = [s for s in rows if q in s["title"].lower()]
        if hits:
```

```python
def dispatch_panel(action):
    """Turn the panel's one Action into the command it names.

    Every branch goes through the same entry point the menu used, so the panel
    gains no write path of its own — registry.set_headless() keeps the runner
    exclusive, registry.set_field() keeps display and colour validated, and
    accounts/launch keep the invariants they already hold.
    """
    if action is None:
        return 1
    kind, slug, value = action
    if kind == "switch":
        return SWITCH
    if kind == "new":
        return launch.run(slug, "new", value, True, value)
    if kind == "resume":
        return launch.run(slug, "resume", value, True, None)
    if kind == "headless":
        reg = registry.load()
        _toggle_headless(reg, slug)
        return 0
    if kind == "dangerous":
        return _toggle_dangerous(registry.load(), slug)
    if kind == "hide_icon":
        account = registry.find(registry.load(), slug)
        return _mutate(slug, "hide_icon", not account["hide_icon"])
    if kind in ("display", "color"):
        return _mutate(slug, kind, value)
    if kind == "add":
        return cmd_add(True)
    if kind in ("rename", "remove"):
        return cmd_manage(kind, slug, True)
    raise ValueError(f"unknown panel action: {kind}")
```

```python
`gi` is imported inside show(), never at module scope. ccs statusline runs in
every prompt of every session and ccs -p is the passthrough; neither may be
breakable by a missing PyGObject, so importing this module must stay free.

MISSING_DEPS = ("ccs: the panel needs PyGObject, GTK4 and gtk4-layer-shell — "
                "install them with: sudo pacman -S python-gobject gtk4 "

def _notify(message: str) -> None:
    """Say it where a bar click can be seen. stderr goes to Waybar's log,
    which nobody is reading at the moment they click something."""
    if shutil.which("notify-send"):
        subprocess.run(["notify-send", "CCAS", message], check=False)
    print(message, file=__import__("sys").stderr)


def _load_css(display) -> None:
    from gi.repository import Gdk, Gtk
    path = paths.menu_css()
    if not path.exists():
        return
    provider = Gtk.CssProvider()
    provider.load_from_path(str(path))
    Gtk.StyleContext.add_provider_for_display(
        display, provider, Gtk.STYLE_PROVIDER_PRIORITY_USER)


def show(state: dict):
    """Open the panel, block, return the chosen Action or None."""
    try:
        import gi
        gi.require_version("Gtk", "4.0")
        gi.require_version("Gtk4LayerShell", "1.0")
        from gi.repository import Gtk, Gtk4LayerShell as LayerShell
    except (ImportError, ValueError):
        _notify(MISSING_DEPS)
        return None

    chosen = {"action": None}
    app = Gtk.Application(application_id="dev.ccas.panel")

    def on_activate(_app):
        window = Gtk.ApplicationWindow(application=app)
        window.set_default_size(WIDTH, HEIGHT)
        window.add_css_class("ccas-panel")

        LayerShell.init_for_window(window)
        LayerShell.set_layer(window, LayerShell.Layer.OVERLAY)
        LayerShell.set_anchor(window, LayerShell.Edge.TOP, True)
        LayerShell.set_margin(window, LayerShell.Edge.TOP, 8)
        # ON_DEMAND, not EXCLUSIVE: the panel takes the keyboard while focused
        # without locking the compositor out if it somehow fails to close.
        LayerShell.set_keyboard_mode(window,
                                     LayerShell.KeyboardMode.ON_DEMAND)

        _load_css(window.get_display())

        esc = Gtk.EventControllerKey()
        esc.connect("key-pressed", lambda _c, keyval, *_a:
                    (window.close(), True)[1] if keyval == 0xff1b else False)
        window.add_controller(esc)

        # Click-outside: a layer surface loses focus when the pointer goes
        # elsewhere, which is the dismissal a menu is expected to have.
        focus = Gtk.EventControllerFocus()
        focus.connect("leave", lambda _c: window.close())
        window.add_controller(focus)

        window.set_child(build_body(state, chosen, window))
        window.present()

    app.connect("activate", on_activate)
    app.run([])
    return chosen["action"]


def build_body(state, chosen, window):
    """Filled in by Tasks 7 and 8. Returns the panel's root widget."""
    from gi.repository import Gtk
    return Gtk.Box(orientation=Gtk.Orientation.VERTICAL)
```
